## ArrayStack 구현
```py
class ArrayStack:
    def __init__(self, capacity = 100):
        self.capacity = capacity
        self.array = [None] * capacity
        self.top = -1  

    # top = -1 : 스택이 공백
    # capacity - 1 : 스택이 가득 참

    #공백 검사
    def isEmpty(self):
        return self.top == -1
    
    #포화 검사
    def isFull(self):
        return self.top == self.capacity - 1
    

    #push(e)
    def push(self, e):
        if not self.isFull():
            self.top += 1
            self.array[self.top] = e   
        else:
            print("overflow")
            exit()
        
    #pop(e)
    def pop(self):
        if not self.isEmpty():
            result = self.array[self.top] 
            self.top -= 1

            return result
        else:
            print("underflow")
            exit()

    #peek()
    def peek(self):
        if not self.isEmpty():
            return self.array[self.top]
        else:
            print("isEmpty")
            exit()


    def __str__(self):
        return str(self.array[0:self.top + 1])
    


# 문자열 역순 출력
stack = ArrayStack()
arr = input("입력 : ")
for i in arr:
    stack.push(i)

print(stack)
for _ in range(4):
    print(stack.pop(), end='')


```

```py
#checkBrackets
def checkBrackets(statement):
    stack = ArrayStack()
    for ch in statement:
        if ch == "(" or ch == "{" or ch == "[":
            stack.push(ch)
    
        elif ch == ")" or ch == "}" or ch == "]":
            if stack.isEmpty(): return False
            else:
                left = stack.pop()
                if (ch == ")" and left != "(") or \
                    (ch == "}" and left != "{") or \
                    (ch == "]" and left != "["):
                    return False

    return stack.isEmpty()  #비어있다면 True 반환
```

## 후위표기식을 스택을 이용해 계산
- 수식의 왼쪽부터 시작
- 피연산자가 나오면 스택에 저장
- 연산자가 나오면 스택에서 피연산자 두개를 꺼내 연산
- 연산 결과를 스택에 저장
- 이후, 스택에 남은 값이 최종결과

```py
   
#후위 표기식을 리스트에 저장한다.
def evalPostfix(expr): #expr = ['8', '4', '+', '3' ...]
    s = ArrayStack()

    for token in expr:
        if token in "+-*/":
            val2 = s.pop()
            val1 = s.pop()
            if token == '+': s.push(val1 + val2)
            elif (token == '-'): s.push(val1 - val2)
            elif (token == '*'): s.push(val1 * val2)
            elif (token == '/'): s.push(val1 / val2)
        else:
            s.push(float(token))

    return s.pop()  #최종값 pop해서 반환

#main
expr1 = list("82/3-32*+")

print(evalPostfix(expr1))
```

## 중위 표기 수식을 후위표기 수식으로 변환
 - 변환 알고리즘의 아이디어
      1. 피연산자는 바로 출력
      2. 연산자는 stack에 저장 (연산자 우선순위가 높을수록 peek가 높다)
      3. 괄호는 특별 처리
            - 왼쪽 괄호는 스택에 삽입
            - 오른쪽 괄호가 나오면 왼쪽 괄호가 나올 때 까지 pop()하여 연산자 출력

```py
# 피연산자는 바로 출력, 연산자는 스택에 삽입
# 연산자 우선순위를 지켜 삽입(우선순위가 높은 것이 스택 상단으로) -> 만일 우선순위가 안지켜질 시 연산자를 pop해 출력 다음 연산자는 스택삽입
# 왼쪽괄호는 무조건 삽입, 오른쪽 괄호 등장 시 왼쪽 괄호가 나올 때까지 스택 내 연산자 모두 pop
# 스택 끝 도달 시 스택 내 요소 전부 pop


#연산자 우선순위 정하기(스택에서 우선순위가 높은게 스택 상단에 위치해야함)
def precedence(op):
    if op == '(' or op == ')': return 0         #연산자 우선순위 가장 낮음
    elif op == '+' or op == '-': return 1
    elif op == '*' or op == '/': return 2       #연산자 우선순위 가장 높음
    else: return -1

def Infix2Postfix(expr):
    s = ArrayStack()
    output = []

    for term in expr:
        #왼쪽 괄호는 바로 삽입
        if term == '(':
            s.push(term)

        #오른쪽 괄호는 왼쪽 괄호가 나올 때까지 pop
        elif term == ')':
            while not s.isEmpty():
                op = s.pop()
                if op == '(': break
                else: output.append(op)
        
        #연산자 우선순위 규칙
        elif term in "+-*/":
            while not s.isEmpty():
                op = s.peek()
                if (precedence(term) <= precedence(op)):
                    output.append(op)
                    s.pop()
                else: break
            s.push(term)

        #피연산자면 출력
        else:
            output.append(term)
    
    #스택에 남은 연산자들 전부 pop
    while not s.isEmpty():
        output.append(s.pop())

    return output
```
